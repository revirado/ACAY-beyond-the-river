    28→    28→
    29→    29→function circle(ctx: C, cx: number, cy: number, r: number, color: string): void {
    30→    30→  ctx.beginPath()
    31→    31→  ctx.arc(cx, cy, r, 0, Math.PI * 2)
    32→    32→  ctx.fillStyle = color
    33→    33→  ctx.fill()
    34→    34→}
    35→    35→
    36→    36→function poly(ctx: C, pts: Array<[number, number]>, color: string): void {
    37→    37→  ctx.beginPath()
    38→    38→  pts.forEach(([x, y], i) => (i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)))
    39→    39→  ctx.closePath()
    40→    40→  ctx.fillStyle = color
    41→    41→  ctx.fill()
    42→    42→}
    43→    43→
    44→    44→function rect(ctx: C, x: number, y: number, w: number, h: number, color: string): void {
    45→    45→  ctx.fillStyle = color
    46→    46→  ctx.fillRect(x, y, w, h)
    47→    47→}
