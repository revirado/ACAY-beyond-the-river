    27→/* ────────────────────── Helpers ────────────────────── */
    28→
    29→function circle(ctx: C, cx: number, cy: number, r: number, color: string): void {
    30→  ctx.beginPath()
    31→  ctx.arc(cx, cy, r, 0, Math.PI * 2)
    32→  ctx.fillStyle = color
    33→  ctx.fill()
    34→}
    35→
    36→function poly(ctx: C, pts: Array<[number, number]>, color: string): void {
    37→  ctx.beginPath()
    38→  pts.forEach(([x, y], i) => (i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y)))
    39→  ctx.closePath()
    40→  ctx.fillStyle = color
    41→  ctx.fill()
    42→}
    43→
    44→function rect(ctx: C, x: number, y: number, w: number, h: number, color: string): void {
    45→  ctx.fillStyle = color
    46→  ctx.fillRect(x, y, w, h)
    47→}

   212→const rhythmTrack: DrawerDef = {
   213→  w: 1280,
   214→  h: 720,
   215→  draw(ctx: C, w: number, h: number): void {
   216→    const bg = ctx.createLinearGradient(0, 0, 0, h)
   217→    bg.addColorStop(0, '#2a1a10')
   218→    bg.addColorStop(1, '#0f0a08')
   219→    ctx.fillStyle = bg
   220→    ctx.fillRect(0, 0, w, h)
   221→
   222→    const lanes = 3
   223→    const topY = h * 0.15
   224→    const botY = h * 0.92
   225→    const topHalfW = w * 0.12
   226→    const botHalfW = w * 0.42
   227→    const cx = w / 2
   228→
   229→    ctx.save()
   230→    ctx.beginPath()
   231→    ctx.moveTo(cx - topHalfW, topY)
   232→    ctx.lineTo(cx + topHalfW, topY)
   233→    ctx.lineTo(cx + botHalfW, botY)
   234→    ctx.lineTo(cx - botHalfW, botY)
   235→    ctx.closePath()
   236→    const trackGrad = ctx.createLinearGradient(0, topY, 0, botY)
   237→    trackGrad.addColorStop(0, PALETTE.laneBorde)
   238→    trackGrad.addColorStop(1, PALETTE.lane)
   239→    ctx.fillStyle = trackGrad
   240→    ctx.fill()
   241→    ctx.restore()
   242→
   243→    ctx.save()
   244→    ctx.strokeStyle = 'rgba(230,182,85,0.3)'
   245→    ctx.lineWidth = 2
   246→    for (let i = 0; i <= lanes; i++) {
   247→      const t = i / lanes
   248→      const topX = cx - topHalfW + t * topHalfW * 2
   249→      const botX = cx - botHalfW + t * botHalfW * 2
   250→      ctx.beginPath()
   251→      ctx.moveTo(topX, topY)
   252→      ctx.lineTo(botX, botY)
   253→      ctx.stroke()
   254→    }
   255→    ctx.restore()
   256→
   257→    ctx.save()
   258→    ctx.strokeStyle = PALETTE.acento
   259→    ctx.lineWidth = 3
   260→    ctx.globalAlpha = 0.6
   261→    const hitY = botY - 30
   262→    const hitHalfW = botHalfW * 0.92
   263→    ctx.beginPath()
   264→    ctx.moveTo(cx - hitHalfW, hitY)
   265→    ctx.lineTo(cx + hitHalfW, hitY)
   266→    ctx.stroke()
   267→    ctx.restore()
   268→  },
   269→}
